                  <div className="absolute inset-0.5 rounded-full border border-slate-800/60" />
                  
                  {/* Rotating Inner Rose */}

                      if (activeProperty && onUpdatePropertyOffset) {

           =================================================== */}

          <aside

              rightPanelPinned

            }`}
          >

              <div className="flex items-center justify-between border-b border-slate-800/60 pb-2.5">
