                {isVastuChakraActive && (
                  <div 
                    className={`absolute z-40 ${!chakraState.isLocked ? "pointer-events-auto" : "pointer-events-none"}`}
                    style={{ 
                       left: `${chakraState.x}px`, 
                       top: `${chakraState.y}px`,
                       width: '0px',
                       height: '0px'
                     }}
                  >
                    {/* The Scaled & Rotated Container */}
                    <div 
                      className="absolute"
                      style={{
                        transform: `translate(-50%, -50%) rotate(${chakraState.rotation}deg) scale(${chakraState.scale})`,
                        width: '0px',
                        height: '0px'
                      }}
                    >
                      {/* The Hit Area - Exactly matching the Chakra's 840px diameter. 
                          CRITICAL: clipPath ensures ONLY the circle is clickable, 
                          clicks outside fall through to the canvas for panning. */}
                      <div 
                        className="absolute rounded-full"
                        style={{ 
                           width: '840px', 
                           height: '840px', 
                           transform: 'translate(-50%, -50%)',
                           clipPath: 'circle(50% at 50% 50%)',
                           pointerEvents: chakraState.isLocked ? 'none' : 'auto',
                           cursor: chakraState.isLocked ? 'default' : 'move', 
                           backgroundColor: 'rgba(0,0,0,0.001)' // Transparent but catches events
                        }}
                        onPointerDown={(e) => {
                          if (chakraState.isLocked) return;
                          if ((e.target as HTMLElement).closest('.cad-handle')) return;
                          
                          e.stopPropagation();
                          e.currentTarget.setPointerCapture(e.pointerId);
                          setSelectedId("chakra");
                          setIsDraggingChakra(true);
                          const coords = getCanvasCoords(e);
                          setChakraDragOffset({
                            x: coords.x - chakraState.x,
                            y: coords.y - chakraState.y
                          });
                        }}
                      >
                        {/* 45-55% Opacity on the original vibrant colors */}
                        <div className="absolute top-1/2 left-1/2 transition-opacity duration-300" style={{ transform: 'translate(-50%, -50%)', pointerEvents: 'none', opacity: selectedId === "chakra" ? 0.55 : 0.45 }}>
                          <VastuChakraSVGOverlay className="pointer-events-none" />
                        </div>
                      </div>

                      {/* Handles Container */}
                      {!chakraState.isLocked && selectedId === "chakra" && (
                        <div className="absolute pointer-events-none" style={{ width: '0px', height: '0px' }}>
                          
                           {/* Rotation Handle (Attached directly above the Chakra, 25px away) */}
                           <div className="absolute flex flex-col items-center pointer-events-none" style={{ transform: `translate(-50%, -100%) translateY(-420px)` }}>
                             {/* Rotation Handle */}
                             <div 
                               className="bg-slate-900 border border-emerald-500 rounded-full flex items-center justify-center shadow-lg hover:bg-emerald-500/20 transition-colors pointer-events-auto cursor-grab cad-handle"
                               style={{
                                 width: `${20 / chakraState.scale}px`,
                                 height: `${20 / chakraState.scale}px`,
                                 borderWidth: `${1.5 / chakraState.scale}px`
                               }}
                               onPointerDown={(e) => {
                                 e.stopPropagation();
                                 e.currentTarget.setPointerCapture(e.pointerId);
                                 setIsRotatingChakra(true);
                                 const coords = getCanvasCoords(e);
                                 const dx = coords.x - chakraState.x;
                                 const dy = coords.y - chakraState.y;
                                 setChakraInteractionStart({
                                   angle: (Math.atan2(dy, dx) * 180) / Math.PI,
                                   rotation: chakraState.rotation,
                                   x: 0,
                                   y: 0,
                                   scale: chakraState.scale
                                 });
                               }}
                               title="Rotate Vastu Chakra"
                             >
                               <svg style={{ width: `${12 / chakraState.scale}px`, height: `${12 / chakraState.scale}px` }} className="text-emerald-400 pointer-events-none" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                                 <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15" />
                               </svg>
                             </div>
                             {/* Thin line connecting handle to Chakra */}
                             <div className="w-[1px] bg-emerald-500" style={{ height: `${25 / chakraState.scale}px`, opacity: 0.6 }} />
                           </div>

                           {/* Resize Handle (Attached directly right of the Chakra, 25px away) */}
                           <div className="absolute flex flex-row items-center pointer-events-none" style={{ transform: `translate(0%, -50%) translateX(420px)` }}>
                             {/* Thin line connecting handle to Chakra */}
                             <div className="h-[1px] bg-emerald-500" style={{ width: `${25 / chakraState.scale}px`, opacity: 0.6 }} />
                             
                             {/* Resize Handle */}
                             <div 
                               className="bg-slate-900 border border-emerald-500 rounded-full flex items-center justify-center shadow-lg hover:bg-emerald-500/20 transition-colors pointer-events-auto cursor-ew-resize cad-handle"
                               style={{
                                 width: `${20 / chakraState.scale}px`,
                                 height: `${20 / chakraState.scale}px`,
                                 borderWidth: `${1.5 / chakraState.scale}px`
                               }}
                               onPointerDown={(e) => {
                                 e.stopPropagation();
                                 e.currentTarget.setPointerCapture(e.pointerId);
                                 setIsResizingChakra(true);
                                 const coords = getCanvasCoords(e);
                                 const dx = coords.x - chakraState.x;
                                 const dy = coords.y - chakraState.y;
                                 setChakraResizeStart({
                                   scale: chakraState.scale,
                                   dist: Math.sqrt(dx * dx + dy * dy)
                                 });
                               }}
                               title="Scale Vastu Chakra"
                             >
                               <svg style={{ width: `${12 / chakraState.scale}px`, height: `${12 / chakraState.scale}px` }} className="text-emerald-400 pointer-events-none" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                                 <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4 8V4m0 0h4M4 4l5 5m11-1V4m0 0h-4m4 0l-5 5M4 16v4m0 0h4m-4 0l5-5m11 5l-5-5m5 5v-4m0 4h-4" />
                               </svg>
                             </div>
                           </div>

                        </div>
